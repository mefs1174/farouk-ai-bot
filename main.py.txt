@bot.message_handler(func=lambda message: str(message.chat.id) == str(ADMIN_CHAT_ID) and message.text == "📢 إرسال إشعار لجميع المستخدمين")
def broadcast_prompt(message):
    msg = bot.send_message(message.chat.id, "✍️ اكتب رسالة الإشعار التي تريد إرسالها لجميع العملاء الآن:")
    bot.register_next_step_handler(msg, execute_broadcast)

def execute_broadcast(message):
    chat_id = message.chat.id
    if str(chat_id) != str(ADMIN_CHAT_ID):
        return
        
    broadcast_text = message.text
    success_count = 0
    fail_count = 0
    
    bot.send_message(chat_id, f"⏳ جاري بدء البث إلى {len(all_users_set)} مستخدم مسجل...")
    
    for uid in all_users_set:
        if str(uid) == str(ADMIN_CHAT_ID):
            continue
        try:
            bot.send_message(uid, f"📢 **إشعار هام من إدارة Farouk AI Studio:**\n\n{broadcast_text}", parse_mode="Markdown")
            success_count += 1
            time.sleep(0.5)  # مهلة زمنية آمنة لمنع حظر تليجرام
        except Exception as e:
            print(f"Failed to send to {uid}: {e}")
            fail_count += 1
            
    bot.send_message(chat_id, f"✅ **تم الانتهاء من الإرسال بنجاح!**\n- تم الإرسال لـ: {success_count} عميل\n- فشل الإرسال لـ: {fail_count}", parse_mode="Markdown", reply_markup=admin_control_keyboard())

@bot.callback_query_handler(func=lambda call: call.data.startswith("reply_to_"))
def ask_admin_reply(call):
    chat_id = call.message.chat.id
    if str(chat_id) != str(ADMIN_CHAT_ID):
        return
    client_id = call.data.split("_")[2]
    admin_reply_targets[chat_id] = client_id
    msg = bot.send_message(chat_id, f"✍️ اكتب رسالتك الآن للعميل (رقم العميل: `{client_id}`):", parse_mode="Markdown")
    bot.register_next_step_handler(msg, send_message_to_client)

def send_message_to_client(message):
    chat_id = message.chat.id
    if str(chat_id) != str(ADMIN_CHAT_ID):
        return
    client_id = admin_reply_targets.get(chat_id)
    reply_text = message.text
    if client_id:
        try:
            bot.send_message(client_id, f"📢 **رد من إدارة Farouk AI Studio:**\n\n{reply_text}")
            bot.send_message(chat_id, f"✅ تم إرسال الرد بنجاح إلى العميل ذو الـ ID: `{client_id}`", parse_mode="Markdown")
        except Exception as e:
            bot.send_message(chat_id, f"❌ حدث خطأ أثناء إرسال الرسالة:\n`{e}`", parse_mode="Markdown")

    bot.register_next_step_handler(msg, get_phone)

def get_phone(message):
    chat_id = message.chat.id
    phone = message.text.strip()
    
    if not (phone.isdigit() and phone.startswith("010") and len(phone) == 11):
        msg = bot.send_message(chat_id, "❌ **خطأ في الرقم!** يجب أن يتكون رقم المحول منه من 11 رقماً ويبدأ بـ **010**.\nمن فضلك اكتب الرقم بشكل صحيح مرة أخرى:")
        bot.register_next_step_handler(msg, get_phone)
        return

    if chat_id not in user_temp_data:
        user_temp_data[chat_id] = {}
    user_temp_data[chat_id]["phone"] = phone
    
    data = user_temp_data[chat_id]
    user_name = message.from_user.first_name
    username = f"@{message.from_user.username}" if message.from_user.username else "لا يوجد"
    
    bot.reply_to(message, "✅ تم استلام طلبك وتحقق من الرقم بنجاح! جاري المراجعة وتنفيذ فوراً.")
    
    admin_text = (
        "🚨 **طلب جديد مدفوع!**\n\n"
        f"👤 العميل: {user_name} ({username})\n"
        f"🆔 الآي دي: `{chat_id}`\n"
        f"📌 الخدمة: {data.get('service')}\n"
        f"💰 السعر المقرر: {data.get('price')}\n"
        f"💵 المبلغ المحول: {data.get('amount')}\n"
        f"📱 رقم المحول منه: {phone}"
    )
    
    admin_markup = InlineKeyboardMarkup()
    admin_markup.add(InlineKeyboardButton("💬 رد على العميل", callback_data=f"reply_to_{chat_id}"))
    
    try:
        bot.send_message(ADMIN_CHAT_ID, admin_text, parse_mode="Markdown", reply_markup=admin_markup)
    except Exception as e:
        print(f"Admin error: {e}")
