import telebot
from telebot.types import InlineKeyboardMarkup, InlineKeyboardButton

bot = telebot.TeleBot(BOT_TOKEN)

# --- القائمة الرئيسية ---
def main_menu_keyboard():
    markup = InlineKeyboardMarkup(row_width=2)
    markup.add(
        InlineKeyboardButton("💎 اشتراكات البرامج والمواقع", callback_data="sub_menu"),
        InlineKeyboardButton("📈 تزويد المتابعين والتفاعل", callback_data="smm_menu")
    )
    markup.add(
        InlineKeyboardButton("🚀 باقات إدارة السوشيال ميديا", callback_data="management_menu"),
        InlineKeyboardButton("🎨 التصميم، المواقع والفيديوهات", callback_data="design_menu")
    )
    markup.add(
        InlineKeyboardButton("👤 حسابي وطريقة الدفع", callback_data="account_menu"),
        InlineKeyboardButton("📞 الدعم الفني وتواصل معنا", callback_data="support")
    )
    return markup

@bot.message_handler(commands=['start'])
def send_welcome(message):
    welcome_text = (
        "أهلاً بيك في **Farouk AI Studio** 🚀\n"
        "وكالتك الشاملة لخدمات السوشيال ميديا، الذكاء الاصطناعي، وتطوير الأعمال.\n\n"
        "اختر الخدمة المطلوبة من القائمة أدناه:"
    )
    bot.send_message(message.chat.id, welcome_text, parse_mode="Markdown", reply_markup=main_menu_keyboard())
